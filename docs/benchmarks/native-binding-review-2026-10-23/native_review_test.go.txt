var nativeReviewRawFactory func(int, int, int, int, int) (codecEncoder, int, error)

// This measurement-only file is overlaid into avc; no production source changes.
func TestNativeReviewEncode(t *testing.T) {
	if os.Getenv("COMPARE_RUN") == "" {
		t.Skip("comparison not requested")
	}
	const fps = 60
	width, height := nativeReviewInt(t, "COMPARE_WIDTH", 1920), nativeReviewInt(t, "COMPARE_HEIGHT", 1080)
	qp, requested := nativeReviewInt(t, "COMPARE_QP", 20), nativeReviewInt(t, "COMPARE_SLICES", 1)
	count, warmup := nativeReviewInt(t, "COMPARE_FRAMES", 120), nativeReviewInt(t, "COMPARE_WARMUP", 16)
	if width < 2 || height < 2 || width%2 != 0 || height%2 != 0 || qp < 0 || qp > 51 || (requested != 1 && requested != 4) || count < 1 || warmup < 0 {
		t.Fatal("invalid comparison configuration")
	}
	data, err := os.ReadFile(os.Getenv("COMPARE_I420"))
	if err != nil {
		t.Fatal(err)
	}
	frameBytes := width * height * 3 / 2
	if len(data) == 0 || len(data)%frameBytes != 0 {
		t.Fatal("invalid I420 input length")
	}
	inputFrames := len(data) / frameBytes
	hash := sha256.Sum256(data)
	mbCount := ((width + 15) / 16) * ((height + 15) / 16)
	var masks [][]bool
	if name := os.Getenv("COMPARE_HINTS"); name != "" {
		raw, err := os.ReadFile(name)
		if err != nil {
			t.Fatal(err)
		}
		if len(raw) != inputFrames*mbCount {
			t.Fatalf("hint length %d; want %d", len(raw), inputFrames*mbCount)
		}
		masks = make([][]bool, inputFrames)
		for frame := range masks {
			masks[frame] = make([]bool, mbCount)
			for mb := range masks[frame] {
				value := raw[frame*mbCount+mb]
				if value > 1 {
					t.Fatal("invalid hint")
				}
				masks[frame][mb] = value == 1
			}
		}
	}
	boundary := os.Getenv("COMPARE_BOUNDARY")
	if boundary == "" {
		boundary = "adapter"
	}
	factory := newCodecEncoder
	if boundary == "raw_binding" {
		if nativeReviewRawFactory == nil {
			t.Fatal("raw_binding requires cgo")
		}
		factory = nativeReviewRawFactory
	} else if boundary != "adapter" {
		t.Fatal("invalid COMPARE_BOUNDARY")
	}
	codec, actual, err := factory(width, height, requested, fps, qp)
	if err != nil {
		t.Fatal(err)
	}
	defer codec.close()
	encoder := codec.(damageCodecEncoder)
	timings, cpuTimings := make([]int64, count), make([]int64, count)
	frameSizes := make([]int, count)
	var saved [][]byte
	if os.Getenv("COMPARE_STREAM") != "" {
		saved = make([][]byte, 0, warmup+count)
	}
	encode := func(i int) []byte {
		index := i % inputFrames
		var unchanged []bool
		if masks != nil && i > 0 {
			unchanged = masks[index]
		}
		encoded, err := encoder.encodeDamage(data[index*frameBytes:(index+1)*frameBytes], i == 0, int64(i)*1000/fps, unchanged)
		if err != nil || len(encoded) == 0 {
			t.Fatalf("frame %d: len=%d err=%v", i, len(encoded), err)
		}
		return encoded
	}
	for i := 0; i < warmup; i++ {
		encoded := encode(i)
		if saved != nil {
			saved = append(saved, encoded)
		}
	}
	runtime.GC()
	var memBefore, memAfter runtime.MemStats
	runtime.ReadMemStats(&memBefore)
	var cpuStart, cpuEnd unix.Timespec
	for n := 0; n < count; n++ {
		if err := unix.ClockGettime(unix.CLOCK_PROCESS_CPUTIME_ID, &cpuStart); err != nil {
			t.Fatal(err)
		}
		start := time.Now()
		encoded := encode(warmup + n)
		timings[n] = time.Since(start).Nanoseconds()
		if err := unix.ClockGettime(unix.CLOCK_PROCESS_CPUTIME_ID, &cpuEnd); err != nil {
			t.Fatal(err)
		}
		cpuTimings[n] = cpuEnd.Nano() - cpuStart.Nano()
		frameSizes[n] = len(encoded)
		if saved != nil {
			saved = append(saved, encoded)
		}
	}
	runtime.ReadMemStats(&memAfter)
	result := nativeReviewSummary(timings, cpuTimings, frameSizes, &memBefore, &memAfter)
	result["kind"], result["implementation"], result["case"] = "encode", os.Getenv("COMPARE_IMPL"), os.Getenv("COMPARE_CASE")
	result["boundary"] = boundary
	result["width"], result["height"], result["qp"], result["requested_slices"], result["actual_slices"] = width, height, qp, requested, actual
	result["warmup_frames"], result["input_frames"], result["hint_frames"], result["input_sha256"] = warmup, inputFrames, len(masks), hex.EncodeToString(hash[:])
	result["capture"], result["go_version"], result["gomaxprocs"] = saved != nil, runtime.Version(), runtime.GOMAXPROCS(0)
	nativeReviewPrint(t, result)
	fmt.Printf("BenchmarkNativeReview/%s/QP%d/Requested%d/Actual%d-%d %d %.3f ns/op %.3f cpu_ns/frame %.3f bytes/frame %d p95_ns/frame %d p99_ns/frame %d max_ns/frame %.3f B/op %.3f allocs/op\n", os.Getenv("COMPARE_CASE"), qp, requested, actual, runtime.GOMAXPROCS(0), count, result["mean_ns"], result["mean_cpu_ns"], result["bytes_per_frame"], result["p95_ns"], result["p99_ns"], result["max_ns"], result["go_alloc_bytes_per_frame"], result["go_allocs_per_frame"])
	if name := os.Getenv("COMPARE_DURATIONS"); name != "" {
		var log strings.Builder
		log.WriteString("frame,input_frame,wall_ns,cpu_ns,bytes\n")
		for n, elapsed := range timings {
			fmt.Fprintf(&log, "%d,%d,%d,%d,%d\n", warmup+n, (warmup+n)%inputFrames, elapsed, cpuTimings[n], frameSizes[n])
		}
		if err := os.WriteFile(name, []byte(log.String()), 0600); err != nil {
			t.Fatal(err)
		}
	}
	if name := os.Getenv("COMPARE_STREAM"); name != "" {
		file, err := os.OpenFile(name, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0600)
		if err != nil {
			t.Fatal(err)
		}
		for _, frame := range saved {
			if _, err := file.Write(frame); err != nil {
				file.Close()
				t.Fatal(err)
			}
		}
		if err := file.Close(); err != nil {
			t.Fatal(err)
		}
	}
	runtime.KeepAlive(data)
	runtime.KeepAlive(masks)
}

// The conversion test calls actual production preparation with a no-op codec.
// Overriding only the effective thread budget isolates its conversion effect.
func TestNativeReviewConversion(t *testing.T) {
	if os.Getenv("COMPARE_RUN") == "" {
		t.Skip("comparison not requested")
	}
	width, height := nativeReviewInt(t, "COMPARE_WIDTH", 1920), nativeReviewInt(t, "COMPARE_HEIGHT", 1080)
	sourceWidth, sourceHeight := nativeReviewInt(t, "COMPARE_SOURCE_WIDTH", width), nativeReviewInt(t, "COMPARE_SOURCE_HEIGHT", height)
	count, warmup := nativeReviewInt(t, "COMPARE_FRAMES", 120), nativeReviewInt(t, "COMPARE_WARMUP", 16)
	if width < 16 || height < 16 || width%2 != 0 || height%2 != 0 || sourceWidth < 2 || sourceHeight < 2 || count < 1 || warmup < 0 {
		t.Fatal("invalid conversion geometry or count")
	}
	imgs := [2]*image.RGBA{image.NewRGBA(image.Rect(0, 0, sourceWidth, sourceHeight)), image.NewRGBA(image.Rect(0, 0, sourceWidth, sourceHeight))}
	for n, img := range imgs {
		for i := range img.Pix {
			img.Pix[i] = byte(i*17 + i/113 + n*53)
		}
	}
	for _, bgrx := range []bool{false, true} {
		for _, threads := range []int{1, 4} {
			format := "rgba"
			if bgrx {
				format = "bgrx"
			}
			e := &Encoder{codec: nativeReviewNoopCodec{}, width: width, height: height, i420: make([]byte, width*height*3/2), threads: threads, backend: BackendSoftware}
			prepare := func(i int) {
				img := imgs[i%len(imgs)]
				var err error
				if bgrx {
					err = e.PrepareBGRXDamage(img.Pix, img.Stride, sourceWidth, sourceHeight, nil, false)
				} else {
					err = e.PrepareDamage(img, nil, false)
				}
				if err != nil {
					t.Fatal(err)
				}
				e.prepared = false
			}
			for i := 0; i < warmup; i++ {
				prepare(i)
			}
			timings, cpuTimings := make([]int64, count), make([]int64, count)
			sizes := make([]int, count)
			runtime.GC()
			var before, after runtime.MemStats
			runtime.ReadMemStats(&before)
			var cs, ce unix.Timespec
			for n := 0; n < count; n++ {
				if err := unix.ClockGettime(unix.CLOCK_PROCESS_CPUTIME_ID, &cs); err != nil {
					t.Fatal(err)
				}
				start := time.Now()
				prepare(warmup + n)
				timings[n] = time.Since(start).Nanoseconds()
				if err := unix.ClockGettime(unix.CLOCK_PROCESS_CPUTIME_ID, &ce); err != nil {
					t.Fatal(err)
				}
				cpuTimings[n] = ce.Nano() - cs.Nano()
			}
			runtime.ReadMemStats(&after)
			result := nativeReviewSummary(timings, cpuTimings, sizes, &before, &after)
			result["kind"], result["format"], result["threads"] = "conversion", format, threads
			result["width"], result["height"], result["source_width"], result["source_height"] = width, height, sourceWidth, sourceHeight
			result["warmup_frames"], result["go_version"], result["gomaxprocs"] = warmup, runtime.Version(), runtime.GOMAXPROCS(0)
			nativeReviewPrint(t, result)
		}
	}
	runtime.KeepAlive(imgs)
}

type nativeReviewNoopCodec struct{}

func (nativeReviewNoopCodec) encode([]byte, bool, int64) ([]byte, error) { return nil, nil }
func (nativeReviewNoopCodec) close()                                     {}

func nativeReviewSummary(wall, cpu []int64, sizes []int, before, after *runtime.MemStats) map[string]any {
	var total, cpuTotal, bytes int64
	var missed60, missed30 int
	for n, v := range wall {
		total += v
		cpuTotal += cpu[n]
		bytes += int64(sizes[n])
		if v > int64(time.Second)/60 {
			missed60++
		}
		if v > int64(time.Second)/30 {
			missed30++
		}
	}
	sorted := slices.Clone(wall)
	slices.Sort(sorted)
	count := len(wall)
	n := float64(count)
	percentile := func(p int) int64 { return sorted[(count*p+99)/100-1] }
	return map[string]any{
		"frames": count, "total_ns": total, "mean_ns": float64(total) / n, "mean_cpu_ns": float64(cpuTotal) / n,
		"p50_ns": percentile(50), "p95_ns": percentile(95), "p99_ns": percentile(99), "max_ns": sorted[count-1],
		"missed_16_67ms": missed60, "missed_33_33ms": missed30, "total_bytes": bytes, "bytes_per_frame": float64(bytes) / n,
		"go_total_alloc_bytes": after.TotalAlloc - before.TotalAlloc, "go_mallocs": after.Mallocs - before.Mallocs, "go_frees": after.Frees - before.Frees,
		"go_alloc_bytes_per_frame": float64(after.TotalAlloc-before.TotalAlloc) / n, "go_allocs_per_frame": float64(after.Mallocs-before.Mallocs) / n,
		"go_num_gc": after.NumGC - before.NumGC, "go_gc_pause_ns": after.PauseTotalNs - before.PauseTotalNs,
		"go_heap_alloc_delta": int64(after.HeapAlloc) - int64(before.HeapAlloc),
	}
}
func nativeReviewInt(t *testing.T, name string, fallback int) int {
	t.Helper()
	value := os.Getenv(name)
	if value == "" {
		return fallback
	}
	n, err := strconv.Atoi(value)
	if err != nil {
		t.Fatalf("%s: %v", name, err)
	}
	return n
}
func nativeReviewPrint(t *testing.T, value any) {
	t.Helper()
	data, err := json.Marshal(value)
	if err != nil {
		t.Fatal(err)
	}
	fmt.Printf("RESULT %s\n", data)
}
