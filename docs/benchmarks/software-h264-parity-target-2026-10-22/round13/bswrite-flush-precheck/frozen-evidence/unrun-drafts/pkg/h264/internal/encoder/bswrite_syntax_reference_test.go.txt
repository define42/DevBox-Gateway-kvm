package encoder

import "github.com/define42/DevBox-Gateway-kvm/pkg/h264/internal/common"

// Frozen round12-next syntax and common bit writers, with names/calls changed
// only. Residual coding is unchanged by this candidate and remains shared.

func bsSyntaxReferenceWRITE_BE_32(ptr []uint8, off int, val uint32) {
	ptr[off+0] = uint8(val >> 24)
	ptr[off+1] = uint8(val >> 16)
	ptr[off+2] = uint8(val >> 8)
	ptr[off+3] = uint8(val >> 0)
}

func bsSyntaxReferenceBsWriteBits(pBitString *common.SBitStringAux, iLen int32, kuiValue uint32) int32 {
	if iLen < pBitString.ILeftBits {
		pBitString.UiCurBits = (pBitString.UiCurBits << uint32(iLen)) | kuiValue
		pBitString.ILeftBits -= iLen
	} else {
		iLen -= pBitString.ILeftBits
		pBitString.UiCurBits = (pBitString.UiCurBits << uint32(pBitString.ILeftBits)) | (kuiValue >> uint32(iLen))
		bsSyntaxReferenceWRITE_BE_32(pBitString.PBuf, pBitString.PCurBuf, pBitString.UiCurBits)
		pBitString.PCurBuf += 4
		pBitString.UiCurBits = kuiValue & ((uint32(1) << uint32(iLen)) - 1)
		pBitString.ILeftBits = 32 - iLen
	}
	return 0
}

func bsSyntaxReferenceBsWriteOneBit(pBitString *common.SBitStringAux, kuiValue uint32) int32 {
	bsSyntaxReferenceBsWriteBits(pBitString, 1, kuiValue)
	return 0
}

func bsSyntaxReferenceBsWriteUE(pBitString *common.SBitStringAux, kuiValue uint32) int32 {
	iTmpValue := kuiValue + 1
	if 256 > kuiValue {
		bsSyntaxReferenceBsWriteBits(pBitString, int32(common.G_kuiGolombUELength[kuiValue]), kuiValue+1)
	} else {
		var n uint32
		if iTmpValue&0xffff0000 != 0 {
			iTmpValue >>= 16
			n += 16
		}
		if iTmpValue&0xff00 != 0 {
			iTmpValue >>= 8
			n += 8
		}

		//n += (g_kuiGolombUELength[iTmpValue] >> 1);

		n += common.G_kuiGolombUELength[iTmpValue-1] >> 1
		bsSyntaxReferenceBsWriteBits(pBitString, int32((n<<1)+1), kuiValue+1)
	}
	return 0
}

func bsSyntaxReferenceBsWriteSE(pBitString *common.SBitStringAux, kiValue int32) int32 {
	var iTmpValue uint32
	if 0 == kiValue {
		bsSyntaxReferenceBsWriteOneBit(pBitString, 1)
	} else if 0 < kiValue {
		iTmpValue = uint32((kiValue << 1) - 1)
		bsSyntaxReferenceBsWriteUE(pBitString, iTmpValue)
	} else {
		iTmpValue = uint32((-kiValue) << 1)
		bsSyntaxReferenceBsWriteUE(pBitString, iTmpValue)
	}
	return 0
}

func bsSyntaxReferenceBsWriteTE(pBs *common.SBitStringAux, kiX int32, kuiValue uint32) {
	if 1 == kiX {
		var uiBit uint32
		if kuiValue == 0 { // !kuiValue
			uiBit = 1
		}
		bsSyntaxReferenceBsWriteOneBit(pBs, uiBit)
	} else {
		bsSyntaxReferenceBsWriteUE(pBs, kuiValue)
	}
}

func bsSyntaxReferenceWelsSpatialWriteMbPred(pEncCtx *sWelsEncCtx, pSlice *SSlice, pCurMb *SMB) {
	pMbCache := &pSlice.sMbCacheInfo
	pBs := pSlice.pSliceBsa
	pSliceHeadExt := &pSlice.sSliceHeaderExt
	iNumRefIdxl0ActiveMinus1 := int32(pSliceHeadExt.sSliceHeader.uiNumRefIdxL0Active) - 1

	uiMbType := pCurMb.uiMbType
	iCbpChroma := int32(pCurMb.uiCbp >> 4)
	iCbpLuma := int32(pCurMb.uiCbp & 15)
	i := 0

	var sMvd [2]SMVUnitXY

	iMbOffset := int32(0)

	switch pSliceHeadExt.sSliceHeader.eSliceType {
	case common.I_SLICE:
		iMbOffset = 0
	case common.P_SLICE:
		iMbOffset = 5
	default:
		return
	}

	switch uiMbType {
	case common.MB_TYPE_INTRA4x4:
		/* mb type */
		bsSyntaxReferenceBsWriteUE(pBs, uint32(iMbOffset+0))

		/* prediction: luma */
		pPredFlag := pMbCache.pPrevIntra4x4PredModeFlag
		pRemMode := pMbCache.pRemIntra4x4PredModeFlag
		for {
			var uiPredFlag uint32
			if pPredFlag[i] {
				uiPredFlag = 1
			}
			bsSyntaxReferenceBsWriteOneBit(pBs, uiPredFlag) /* b_prev_intra4x4_pred_mode */

			if !pPredFlag[i] {
				bsSyntaxReferenceBsWriteBits(pBs, 3, uint32(pRemMode[i]))
			}

			i++
			if !(i < 16) {
				break
			}
		}

		/* prediction: chroma */
		bsSyntaxReferenceBsWriteUE(pBs, uint32(g_kiMapModeIntraChroma[pMbCache.uiChmaI8x8Mode]))

	case common.MB_TYPE_INTRA16x16:
		/* mb type */
		iCbpLumaFlag := int32(0)
		if iCbpLuma != 0 {
			iCbpLumaFlag = 12
		}
		bsSyntaxReferenceBsWriteUE(pBs, uint32(1+iMbOffset+int32(g_kiMapModeI16x16[pMbCache.uiLumaI16x16Mode])+(iCbpChroma<<2)+
			iCbpLumaFlag))

		/* prediction: chroma */
		bsSyntaxReferenceBsWriteUE(pBs, uint32(g_kiMapModeIntraChroma[pMbCache.uiChmaI8x8Mode]))

	case common.MB_TYPE_16x16:
		bsSyntaxReferenceBsWriteUE(pBs, 0) //uiMbType
		sMvd[0].sDeltaMv(pCurMb.sMv[0], pMbCache.sMbMvp[0])

		if iNumRefIdxl0ActiveMinus1 > 0 {
			bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[0]))
		}

		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvX))
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvY))

	case common.MB_TYPE_16x8:
		bsSyntaxReferenceBsWriteUE(pBs, 1) //uiMbType

		sMvd[0].sDeltaMv(pCurMb.sMv[0], pMbCache.sMbMvp[0])
		sMvd[1].sDeltaMv(pCurMb.sMv[8], pMbCache.sMbMvp[1])

		if iNumRefIdxl0ActiveMinus1 > 0 {
			bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[0]))
			bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[2]))
		}
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvX)) //block0
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvY))
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[1].iMvX)) //block1
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[1].iMvY))

	case common.MB_TYPE_8x16:
		bsSyntaxReferenceBsWriteUE(pBs, 2) //uiMbType

		sMvd[0].sDeltaMv(pCurMb.sMv[0], pMbCache.sMbMvp[0])
		sMvd[1].sDeltaMv(pCurMb.sMv[2], pMbCache.sMbMvp[1])

		if iNumRefIdxl0ActiveMinus1 > 0 {
			bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[0]))
			bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[1]))
		}
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvX)) //block0
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[0].iMvY))
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[1].iMvX)) //block1
		bsSyntaxReferenceBsWriteSE(pBs, int32(sMvd[1].iMvY))
	}
}

func bsSyntaxReferenceWelsSpatialWriteSubMbPred(pEncCtx *sWelsEncCtx, pSlice *SSlice, pCurMb *SMB) {
	pMbCache := &pSlice.sMbCacheInfo
	pBs := pSlice.pSliceBsa
	pSliceHeadExt := &pSlice.sSliceHeaderExt

	iNumRefIdxl0ActiveMinus1 := int32(pSliceHeadExt.sSliceHeader.uiNumRefIdxL0Active) - 1

	bSubRef0 := false
	kpScan4 := common.G_kuiMbCountScan4Idx[:]

	/* mb type */
	if pCurMb.pRefIndex[0] == 0 && pCurMb.pRefIndex[1] == 0 && pCurMb.pRefIndex[2] == 0 && pCurMb.pRefIndex[3] == 0 {
		bsSyntaxReferenceBsWriteUE(pBs, 4)
		bSubRef0 = false
	} else {
		bsSyntaxReferenceBsWriteUE(pBs, 3)
		bSubRef0 = true
	}

	//step 1: sub_mb_type
	for i := 0; i < 4; i++ {
		switch uint32(pCurMb.uiSubMbType[i]) {
		case common.SUB_MB_TYPE_8x8:
			bsSyntaxReferenceBsWriteUE(pBs, 0)
		case common.SUB_MB_TYPE_8x4:
			bsSyntaxReferenceBsWriteUE(pBs, 1)
		case common.SUB_MB_TYPE_4x8:
			bsSyntaxReferenceBsWriteUE(pBs, 2)
		case common.SUB_MB_TYPE_4x4:
			bsSyntaxReferenceBsWriteUE(pBs, 3)
		default: //should not enter
		}
	}

	//step 2: get and write uiRefIndex and sMvd
	if iNumRefIdxl0ActiveMinus1 > 0 && bSubRef0 {
		bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[0]))
		bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[1]))
		bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[2]))
		bsSyntaxReferenceBsWriteTE(pBs, iNumRefIdxl0ActiveMinus1, uint32(pCurMb.pRefIndex[3]))
	}

	writeMvd := func(k uint8) {
		bsSyntaxReferenceBsWriteSE(pBs, int32(pCurMb.sMv[k].iMvX)-int32(pMbCache.sMbMvp[k].iMvX))
		bsSyntaxReferenceBsWriteSE(pBs, int32(pCurMb.sMv[k].iMvY)-int32(pMbCache.sMbMvp[k].iMvY))
	}

	//write sMvd
	for i := 0; i < 4; i++ {
		uiSubMbType := uint32(pCurMb.uiSubMbType[i])
		if common.SUB_MB_TYPE_8x8 == uiSubMbType {
			writeMvd(kpScan4[0])
		} else if common.SUB_MB_TYPE_4x4 == uiSubMbType {
			writeMvd(kpScan4[0])
			writeMvd(kpScan4[1])
			writeMvd(kpScan4[2])
			writeMvd(kpScan4[3])
		} else if common.SUB_MB_TYPE_8x4 == uiSubMbType {
			writeMvd(kpScan4[0])
			writeMvd(kpScan4[2])
		} else if common.SUB_MB_TYPE_4x8 == uiSubMbType {
			writeMvd(kpScan4[0])
			writeMvd(kpScan4[1])
		}
		kpScan4 = kpScan4[4:]
	}
}

func bsSyntaxReferenceWelsSpatialWriteMbSyn(pEncCtx *sWelsEncCtx, pSlice *SSlice, pCurMb *SMB) int32 {
	pBs := pSlice.pSliceBsa
	pMbCache := &pSlice.sMbCacheInfo
	kuiChromaQpIndexOffset := pEncCtx.pCurDqLayer.sLayerInfo.pPpsP.uiChromaQpIndexOffset

	if common.IS_SKIP(pCurMb.uiMbType) {
		pCurMb.uiLumaQp = pSlice.uiLastMbQp
		pCurMb.uiChromaQp = common.G_kuiChromaQpTable[common.CLIP3_QP_0_51(int32(pCurMb.uiLumaQp)+int32(kuiChromaQpIndexOffset))]

		pSlice.iMbSkipRun++
		return ENC_RETURN_SUCCESS
	} else {
		if pEncCtx.eSliceType != common.I_SLICE {
			bsSyntaxReferenceBsWriteUE(pBs, uint32(pSlice.iMbSkipRun))
			pSlice.iMbSkipRun = 0
		}
		/* Step 1: write mb type and pred */
		if common.IS_Inter_8x8(pCurMb.uiMbType) {
			bsSyntaxReferenceWelsSpatialWriteSubMbPred(pEncCtx, pSlice, pCurMb)
		} else {
			bsSyntaxReferenceWelsSpatialWriteMbPred(pEncCtx, pSlice, pCurMb)
		}

		/* Step 2: write coded block patern */
		if common.IS_INTRA4x4(pCurMb.uiMbType) {
			bsSyntaxReferenceBsWriteUE(pBs, g_kuiIntra4x4CbpMap[pCurMb.uiCbp])
		} else if !common.IS_INTRA16x16(pCurMb.uiMbType) {
			bsSyntaxReferenceBsWriteUE(pBs, g_kuiInterCbpMap[pCurMb.uiCbp])
		}

		/* Step 3: write QP and residual */
		if pCurMb.uiCbp > 0 || common.IS_INTRA16x16(pCurMb.uiMbType) {
			kiDeltaQp := int32(pCurMb.uiLumaQp) - int32(pSlice.uiLastMbQp)
			pSlice.uiLastMbQp = pCurMb.uiLumaQp

			bsSyntaxReferenceBsWriteSE(pBs, kiDeltaQp)
			if WelsWriteMbResidual(pEncCtx.pFuncList, pMbCache, pCurMb, pBs) != 0 {
				return ENC_RETURN_VLCOVERFLOWFOUND
			}
		} else {
			pCurMb.uiLumaQp = pSlice.uiLastMbQp
			pCurMb.uiChromaQp = common.G_kuiChromaQpTable[common.CLIP3_QP_0_51(int32(pCurMb.uiLumaQp)+
				int32(pEncCtx.pCurDqLayer.sLayerInfo.pPpsP.uiChromaQpIndexOffset))]
		}

		/* Step 4: Check the left buffer */
		return CheckBitstreamBuffer(uint32(pSlice.iSliceIdx), pEncCtx, pBs)
	}
}
