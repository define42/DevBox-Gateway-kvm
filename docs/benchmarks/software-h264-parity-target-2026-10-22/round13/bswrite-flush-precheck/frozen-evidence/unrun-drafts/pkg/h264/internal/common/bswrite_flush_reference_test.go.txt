package common_test

import "github.com/define42/DevBox-Gateway-kvm/pkg/h264/internal/common"

// Frozen round12-next implementation, renamed only. This oracle does not call
// the candidate bit writer or its new slow helper.

func bsFlushReferenceWRITE_BE_32(ptr []uint8, off int, val uint32) {
	ptr[off+0] = uint8(val >> 24)
	ptr[off+1] = uint8(val >> 16)
	ptr[off+2] = uint8(val >> 8)
	ptr[off+3] = uint8(val >> 0)
}

func bsFlushReferenceBsWriteBits(pBitString *common.SBitStringAux, iLen int32, kuiValue uint32) int32 {
	if iLen < pBitString.ILeftBits {
		pBitString.UiCurBits = (pBitString.UiCurBits << uint32(iLen)) | kuiValue
		pBitString.ILeftBits -= iLen
	} else {
		iLen -= pBitString.ILeftBits
		pBitString.UiCurBits = (pBitString.UiCurBits << uint32(pBitString.ILeftBits)) | (kuiValue >> uint32(iLen))
		bsFlushReferenceWRITE_BE_32(pBitString.PBuf, pBitString.PCurBuf, pBitString.UiCurBits)
		pBitString.PCurBuf += 4
		pBitString.UiCurBits = kuiValue & ((uint32(1) << uint32(iLen)) - 1)
		pBitString.ILeftBits = 32 - iLen
	}
	return 0
}

func bsFlushReferenceBsWriteOneBit(pBitString *common.SBitStringAux, kuiValue uint32) int32 {
	bsFlushReferenceBsWriteBits(pBitString, 1, kuiValue)
	return 0
}

func bsFlushReferenceBsWriteUE(pBitString *common.SBitStringAux, kuiValue uint32) int32 {
	iTmpValue := kuiValue + 1
	if 256 > kuiValue {
		bsFlushReferenceBsWriteBits(pBitString, int32(common.G_kuiGolombUELength[kuiValue]), kuiValue+1)
	} else {
		var n uint32
		if iTmpValue&0xffff0000 != 0 {
			iTmpValue >>= 16
			n += 16
		}
		if iTmpValue&0xff00 != 0 {
			iTmpValue >>= 8
			n += 8
		}

		//n += (g_kuiGolombUELength[iTmpValue] >> 1);

		n += common.G_kuiGolombUELength[iTmpValue-1] >> 1
		bsFlushReferenceBsWriteBits(pBitString, int32((n<<1)+1), kuiValue+1)
	}
	return 0
}

func bsFlushReferenceBsWriteSE(pBitString *common.SBitStringAux, kiValue int32) int32 {
	var iTmpValue uint32
	if 0 == kiValue {
		bsFlushReferenceBsWriteOneBit(pBitString, 1)
	} else if 0 < kiValue {
		iTmpValue = uint32((kiValue << 1) - 1)
		bsFlushReferenceBsWriteUE(pBitString, iTmpValue)
	} else {
		iTmpValue = uint32((-kiValue) << 1)
		bsFlushReferenceBsWriteUE(pBitString, iTmpValue)
	}
	return 0
}
