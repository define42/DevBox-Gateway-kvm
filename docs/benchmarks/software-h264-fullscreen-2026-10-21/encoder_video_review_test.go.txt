//go:build linux

package avc

import (
	"crypto/sha256"
	"encoding/binary"
	"fmt"
	"image"
	"testing"
)

func TestFullscreenVideoStreamDigest(t *testing.T) {
	for _, fixture := range []string{"Clip", "Dense"} {
		frames := videoReviewFrames(t, fixture)
		for _, threads := range []int{1, 4} {
			for _, fullDamage := range []bool{true, false} {
				name := fmt.Sprintf("%s/%dSlices/fullDamage=%t", fixture, threads, fullDamage)
				t.Run(name, func(t *testing.T) {
					enc, err := NewEncoderWithOptions(videoReviewWidth, videoReviewHeight, Options{Backend: BackendSoftware, Threads: threads, FrameRate: 30})
					if err != nil {
						t.Fatal(err)
					}
					defer enc.Close()
					var regions []image.Rectangle
					if fullDamage {
						regions = []image.Rectangle{frames[0].Rect}
					}
					digest := sha256.New()
					total := 0
					for frame := range videoReviewWarmup + 60 {
						data, err := enc.EncodeDamage(frames[frame%len(frames)], regions, frame == 0)
						if err != nil || len(data) == 0 {
							t.Fatalf("frame %d: bytes=%d error=%v", frame, len(data), err)
						}
						var size [4]byte
						binary.BigEndian.PutUint32(size[:], uint32(len(data)))
						_, _ = digest.Write(size[:])
						_, _ = digest.Write(data)
						total += len(data)
					}
					t.Logf("VIDEO_DIGEST %s %x bytes=%d", name, digest.Sum(nil), total)
				})
			}
		}
	}
}
